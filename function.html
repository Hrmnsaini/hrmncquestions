Swap numbers using Call by address ?

Solution
#include <stdio.h>
void swap(int *x,int *y)
{
    int z;
    z=*x;
    *x=*y;
    *y=z;
}
void print(int a,int b){
    printf("Original Values:\na=%d \nb=%d \n ",a,b);
}
