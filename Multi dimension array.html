Question 1
Enter the four number from user and print it into matrix
Solution

#include<conio.h>

{

         scanf("%d", &mat[i][j]);
        }

{
 for(j=0;j<2;j++)
{
 scanf("%d",&a[i][j]);
 printf("Enter 4 numbers for second matrix: ");
}}
 for(i=0;i<2;i++)
{
for(j=0;j<2;j++)
{
 scanf("%d",&b[i][j]);
 for(i=0;i<2;i++){
 for(j=0;j<2;j++){
 mul[i][j]=0;
 for(k=0;k<2;k++)
 mul[i][j] += a[i][k]*b[k][j];
}

for(j=0;j<2;j++)
printf("%d\t", mul[i][j]);
