Question 1
Enter the four number from user and print it into matrix
Solution
#include<stdio.h> 
#include<conio.h> 
void main()
{
int mat[2][2], i, j;
clrscr();
printf("Enter 4 numbers: ");
for(i=0;i<2;i++)
{
    for(j=0; j<2; j++)
        {
         scanf("%d", &mat[i][j]);
        }
}
printf("Matrix:\n" );
for(i=0; i<2; i++)
{
    for(j=0; j<2; j++)
    {
        printf("%d\t", mat[i][j]);
    }
        printf("\n");
}
}

Output:

Matrix:

8      0


Question 2
Enter the matrix from and transpose them ?
Solution
#include<stdio.h> 
#include<conio.h> 
void main(){
int mat[2][2], trans[2][2], i, j;
clrscr();
 printf("Enter 4 numbers for 2x2 matrix: ");
    for(i=0;i<2;i++){
        for(j=0;j<2;j++){
            scanf("%d",&mat[i][j]);
            trans[j][i] = mat[i][j];
        }
    }
    printf( "Transpose Matrix:\n" );
    for(i=0;i<2;i++)
    {
        for(j=0;j<2;j++)
        {
            printf("%d\t", trans[i][j]);
        }
        printf("\n");
    }
}

Output: 
Enter 4 numbers for 2x2 matrix: 7 3 8 2     
Transpose Matrix:
7       8
3       2

Question 4
Enter 2 matrix from user and subtract them ?
Solution
    #include<stdio.h> 
	#include<conio.h> 
	void main(){
    int a[2][2], b[2][2], sub[2][2], i, j;
	clrscr();
    printf("Enter 4 numbers for first matrix: ");
    for(i=0;i<2;i++)
        for(j=0;j<2;j++)
            scanf("%d",&a[i][j]);
    printf(" Enter 4 numbers for second matrix:" );
    for(i=0;i<2; i++)
        for(j=0;j<2; j++)
            scanf("%d", &b[i][j]);
    for(i=0;i<2; i++)
        for(j=0;j<2; j++)
            sub[i][j] = a[i][j] - b[i][j];
    printf( "Subtraction Matrix:\n" );
    for(i=0; i<2; i++){
        for(j=0; j<2; j++)
            printf("%d\t", sub[i][j]);
        printf("\n" );
    }
}
