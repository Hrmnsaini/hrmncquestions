Question 11
Enter marks of students and calculate average of student's marks?
Solution
#include<stdio.h>
#include<conio.h> 
void main()
{
int num, i, marks, sum=0;
clrscr();
printf("Enter number of students: ");
scanf("%d", &num);
for(i=1; i<=num; i++)
{
 printf("Enter marks of student %d: ", i);
 scanf("%d",&marks);
 sum= sum+marks;
 }
 printf("Average marks: %.2f", (float)sum/num);
}

Output: 
Enter number of students:1
Enter marks of students:
Average marks


Question 12
Calculate simple interest=(P*R*T)/100
Solution

#include<conio.h> 
void main()
{
clrscr();
float P, R, T, SI;
printf("Enter Principal, Rate, Time: ");
scanf("%f %f %f",&P,&R,&T);

}
