Question 6
Create a file and count the number of words in that file.

Solution
#include
#include
void main()
{
FILE *f;
char name[20], data[50],*x,ch;
int y,count=1;
clrscr();
printf("Enter file name:");
gets(name);
fflush(stdin);
printf("Press Ctrl+S to save the file \n Press enter to insert more data");
f=fopen(name,"w");
if(f==NULL)
printf("File is not created");
else
{
do
{
gets(data);
fprintf(f,"%s",data);
ch=getch();
}while(ch!=19);
printf("File saved");
fclose(f);

f=fopen(name,"r");
do
{
x=fgets(data,100,f);
if(x!=0)
{
for(y=0; data[y]!='\0'; y++)
{
if(data[y]==','||data[y]==' ')
{
count++;
y++;
}
}
}
}
while(x!=0);
fclose(f);
printf("Number of words=%d", count);
}
}

Output: 
Enter file name:file 1
Press Ctrl+S to save the file \n Press enter to insert more data
Enter the data:Harman rani
number of characters in file= 2
