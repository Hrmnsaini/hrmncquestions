Question 17

String palindrome example (using reverse) ?

Solution




#include<stdio.h> 
#include<conio.h> 
void main()
{
char str1[50],str2[50];
int i,l;
clrscr();
printf("Pallindrome are words that spells the same forward and backward for example \nNITIN,NAMAN and MOM\n");
printf("Enter any word to check if it is a pallindrome: ");
gets(str1);
l=strlen(str1);
for(i=0;str1[i];i++){
str2[i]=str1[l-1-i];
}
str2[i]=0;
l=strcmp(str1,str2);
if(l==0)printf("Entered string is a pallindrome");
else printf("Entered string is not a pallindrome");
}








Output:

Enter any word to check if it is a pallidrome:mom
Entered string is pallindrome
